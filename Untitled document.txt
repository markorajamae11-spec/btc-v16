class DataAnalyzer:
    async def fetch_candles(self, interval, limit=100):
        try:
            seconds_per_candle = {"1h": 3600, "15m": 900, "4h": 14400}.get(interval, 3600)
            start_time = int((time.time() - (limit * seconds_per_candle)) * 1000)
            
            async with aiohttp.ClientSession() as session:
                payload = {"type": "candleSnapshot", "req": {"coin": "BTC", "interval": interval, "startTime": start_time}}
                async with session.post("https://api.hyperliquid.xyz/info", json=payload, timeout=10) as resp:
                    if resp.status == 200: return await resp.json()
                    return []
        except Exception as e:
            logging.error(f"Candle Fetch Error ({interval}): {e}")
            return []


    async def get_market_data(self):
        raw_1h = await self.fetch_candles("1h")
        if not raw_1h or len(raw_1h) < 50: return None
        
        closes_1h = [float(c['c']) for c in raw_1h]
        highs_1h = [float(c['h']) for c in raw_1h]
        lows_1h = [float(c['l']) for c in raw_1h]
        
        trs = [max(highs_1h[i] - lows_1h[i], abs(highs_1h[i] - closes_1h[i-1]), abs(lows_1h[i] - closes_1h[i-1])) for i in range(1, len(raw_1h))]
        atr = sum(trs[-14:]) / 14.0
        
        ema50 = sum(closes_1h[:50]) / 50.0
        for price in closes_1h[50:]:
            ema50 = (price - ema50) * (2.0 / 51) + ema50


        raw_15m = await self.fetch_candles("15m")
        rsi = 50
        if len(raw_15m) > 15:
            gains, losses = 0, 0
            for j in range(len(raw_15m)-14, len(raw_15m)):
                change = float(raw_15m[j]['c']) - float(raw_15m[j-1]['c'])
                if change > 0: gains += change
                else: losses -= change
            if losses > 0:
                rs = (gains / 14) / (losses / 14)
                rsi = 100 - (100 / (1 + rs))


        raw_4h = await self.fetch_candles("4h")
        active_ob = None
        if len(raw_4h) > 2:
            for i in range(2, min(30, len(raw_4h))):
                curr, prev = raw_4h[-i], raw_4h[-i-1]
                if float(prev['c']) < float(prev['o']) and float(curr['c']) > float(curr['o']) and float(curr['c']) > float(prev['h']):
                    active_ob = float(prev['l'])
                    break


        return {"price": closes_1h[-1], "atr": atr, "ema50": ema50, "rsi": rsi, "active_ob": active_ob}


    async def get_macro_sentiment(self):
        """V17.0 MACRO RADAR: Pulls Geopolitics, Gold, and Crypto News"""
        macro_data = {"fear_greed": 50, "macro_event": "None", "gold_price": 0.0, "global_risk": "LOW"}
        headlines = []
        
        feeds = [
            "https://cointelegraph.com/rss",
            "https://feeds.a.dj.com/rss/RSSWorldNews.xml", 
            "https://finance.yahoo.com/news/rssindex"      
        ]
        
        try:
            async with aiohttp.ClientSession() as session:
                # 1. Fear & Greed
                try:
                    async with session.get("https://api.alternative.me/fng/", timeout=5) as resp:
                        if resp.status == 200:
                            data = await resp.json()
                            macro_data["fear_greed"] = int(data['data'][0]['value'])
                except: pass
                
                # 2. Gold Price (PAXG)
                try:
                    async with session.get("https://api.binance.com/api/v3/ticker/price?symbol=PAXGUSDT", timeout=5) as resp:
                        if resp.status == 200:
                            data = await resp.json()
                            macro_data["gold_price"] = float(data['price'])
                except: pass


                # 3. Scan News for Black Swans
                danger_keywords = ["WAR", "HACK", "EXPLOIT", "BAN", "SEC", "BANKRUPTCY", "FED", "INFLATION", "CRASH"]
                
                for url in feeds:
                    try:
                        async with session.get(url, timeout=5) as resp:
                            if resp.status == 200:
                                xml_data = await resp.text()
                                root = ET.fromstring(xml_data)
                                for item in root.findall('.//item')[:3]:
                                    title = item.find('title').text
                                    if title:
                                        headlines.append(title)
                                        if any(w in title.upper() for w in danger_keywords):
                                            macro_data["global_risk"] = "HIGH_IMPACT_NEWS"
                                            if "WAR" in title.upper() or "HACK" in title.upper():
                                                macro_data["global_risk"] = "CRITICAL_BLACK_SWAN"
                    except: pass


        except Exception as e:
            logging.warning(f"Macro fetch failed: {e}")
            
        return headlines, macro_data

# 3. OBSIDIAN BRAIN
# ==========================================
class ObsidianBrain:
    def __init__(self):
        self.vault = VAULT_DIR
        self.trades_dir = self.vault / "Trades"
        self.news_dir = self.vault / "Headlines"
        self.skips_dir = self.vault / "Skipped"
        for d in [self.trades_dir, self.news_dir, self.skips_dir]:
            d.mkdir(parents=True, exist_ok=True)
        self.last_skip_time = {} 


    def log_trade(self, trade, exit_price, pnl, balance, reason, macro_data):
        now = datetime.now(timezone.utc)
        filename = self.trades_dir / f"{now.strftime('%Y-%m-%d_%H%M%S')}_CLOSE_{trade['side']}.md"
        tag = "#win" if pnl > 0 else "#loss"
        ob_str = f"{trade.get('ob', 0):.2f}" if trade.get('ob') is not None else "null"
        
        content = f"""---
type: trade
event: CLOSE
side: {trade['side']}
entry: {trade['entry_price']:.2f}
exit: {exit_price:.2f}
pnl: {pnl:.2f}
reason: {reason}
fear_greed: {macro_data['fear_greed']}
gold_price: {macro_data.get('gold_price', 0)}
global_risk: "{macro_data.get('global_risk', 'LOW')}"
macro_event: "{macro_data['macro_event']}"
l2_imbalance: {trade.get('l2_imb', 0):.2f}
rsi_at_entry: {trade.get('rsi', 0):.2f}
ob_hit: {ob_str}
---
# {trade['side']} Closed @ ${exit_price:,.2f}
Tags: #trade {tag} #regime/1H #risk/{macro_data.get('global_risk', 'low').lower()}


## 💰 PnL
- **Reason:** {reason}
- **PnL:** ${pnl:+.2f}
- **Balance:** ${balance:,.2f}
"""
        filename.write_text(content, encoding="utf-8")


    def log_skipped_trade(self, reason, market, l2_imb):
        now = time.time()
        if reason in self.last_skip_time and (now - self.last_skip_time[reason]) < 900:
            return 
        self.last_skip_time[reason] = now


        dt = datetime.now(timezone.utc)
        filename = self.skips_dir / f"{dt.strftime('%Y-%m-%d_%H%M%S')}_SKIP_{reason}.md"
        ob_str = f"{market['active_ob']:.2f}" if market['active_ob'] is not None else "null"
        
        content = f"""---
type: skipped
reason: "{reason}"
l2: {l2_imb:.2f}
rsi: {market['rsi']:.2f}
price: {market['price']:.2f}
ob: {ob_str}
---
# Skipped: {reason}
"""
        filename.write_text(content, encoding="utf-8")


    def log_headline(self, headline, source):
        dt = datetime.now(timezone.utc)
        safe_title = "".join([c if c.isalnum() else "_" for c in headline[:30]])
        filename = self.news_dir / f"{dt.strftime('%Y-%m-%d_%H%M')}_NEWS_{safe_title}.md"
        filename.write_text(f"---\ntype: headline\nsource: {source}\n---\n# [{source}] {headline}\n", encoding="utf-8")

class TelegramCommander:
    def __init__(self, bot_instance):
        self.bot = bot_instance
        self.app = Application.builder().token(TG_BOT_TOKEN).build()
        self.app.add_handler(CommandHandler("start", self.start))
        self.app.add_handler(CommandHandler("status", self.status))
        self.app.add_handler(CommandHandler("macro", self.macro))


    async def start(self, update: Update, context: ContextTypes.DEFAULT_TYPE):
        await update.message.reply_text("🤖 *BTC Agent v17.0 (Macro Radar + Equity Guardian)*\nCircuit Breakers Armed.\nHunting...", parse_mode="Markdown")


    async def status(self, update: Update, context: ContextTypes.DEFAULT_TYPE):
        state = self.bot.state
        if state.get("account_failed"):
            await update.message.reply_text("🚨 *ACCOUNT FAILED.* Bot permanently halted to protect prop firm rules.")
            return
        if state["position"] == 0:
            await update.message.reply_text(f"🟢 No active trade.\nBalance: ${state['balance']:.2f}\nL2 Imb: {self.bot.current_l2_imb:+.2f}\nMood: {self.bot.get_mood()}")
            return
        trade = state["active_trade"]
        msg = f"📊 *ACTIVE {trade['side']}*\nEntry: ${trade['entry_price']:,.2f}\nSL: ${trade['stop_loss']:,.2f}\nTP: ${trade['take_profit']:,.2f}\nL2 at Entry: {trade['l2_imb']:+.2f}"
        await update.message.reply_text(msg, parse_mode="Markdown")


    async def macro(self, update: Update, context: ContextTypes.DEFAULT_TYPE):
        headlines, macro_data = await self.bot.analyzer.get_macro_sentiment()
        msg = f"🌍 *Macro Radar*\nFear & Greed: {macro_data['fear_greed']}\nGold Price: ${macro_data.get('gold_price', 0):,.2f}\nGlobal Risk: {macro_data.get('global_risk', 'LOW')}\nEvent: {macro_data['macro_event']}\n\n"
        for h in headlines[:5]: msg += f"• {h}\n"
        await update.message.reply_text(msg, parse_mode="Markdown")


    async def start_async(self):
        await self.app.initialize()
        await self.app.start()
        await self.app.updater.start_polling()
        
    async def send_alert(self, text):
        await self.app.bot.send_message(chat_id=TG_CHAT_ID, text=text, parse_mode="Markdown")

class AutonomousCloudBot:
    def __init__(self):
        self.analyzer = DataAnalyzer()
        self.brain = ObsidianBrain()
        self.state = self.load_state()
        self.tg = TelegramCommander(self)
        self.current_l2_imb = 0.0
        self.latest_market = None
        self.macro = {"fear_greed": 50, "macro_event": "None", "gold_price": 0.0, "global_risk": "LOW"}
        self.last_news_hour = -1


    def load_state(self):
        try:
            with open(STATE_FILE, 'r') as f: 
                state = json.load(f)
                if "daily_start_balance" not in state:
                    state["daily_start_balance"] = state["balance"]
                    state["last_utc_date"] = datetime.now(timezone.utc).strftime('%Y-%m-%d')
                if "account_failed" not in state: state["account_failed"] = False
                if "daily_halted" not in state: state["daily_halted"] = False
                return state
        except FileNotFoundError:
            return {
                "balance": INITIAL_BALANCE, 
                "daily_start_balance": INITIAL_BALANCE,
                "last_utc_date": datetime.now(timezone.utc).strftime('%Y-%m-%d'),
                "position": 0, 
                "consecutive_wins": 0, 
                "active_trade": {},
                "account_failed": False,
                "daily_halted": False
            }


    def save_state(self):
        tmp_file = STATE_FILE.with_suffix('.tmp')
        with open(tmp_file, 'w') as f: json.dump(self.state, f, indent=2)
        os.replace(tmp_file, STATE_FILE)


    def get_mood(self):
        if self.state["balance"] <= 24500.0: return "🛡️ DEFENSIVE"
        elif self.state["consecutive_wins"] >= 2: return "🔥 AGGRESSIVE"
        elif self.state["balance"] >= 26000.0: return "🚀 COMPOUNDING"
        else: return "⚔️ NEUTRAL"


    async def hyperliquid_ws_loop(self):
        uri = "wss://api.hyperliquid.xyz/ws"
        while True:
            try:
                async with websockets.connect(uri, open_timeout=10, ping_interval=20, ping_timeout=20) as ws:
                    await ws.send(json.dumps({"method": "subscribe", "subscription": {"type": "l2Book", "coin": "BTC"}}))
                    async for message in ws:
                        data = json.loads(message)
                        if data.get("channel") == "l2Book":
                            levels = data["data"]["levels"]
                            bids = levels[0][:L2_DEPTH_LEVELS]
                            asks = levels[1][:L2_DEPTH_LEVELS]
                            bid_vol = sum([float(b['sz']) for b in bids])
                            ask_vol = sum([float(a['sz']) for a in asks])
                            if (bid_vol + ask_vol) > 0:
                                self.current_l2_imb = (bid_vol - ask_vol) / (bid_vol + ask_vol)
            except Exception as e:
                logging.error(f"WS Error: {e}")
                await asyncio.sleep(5)


    async def strategy_loop(self):
        while True:
            try:
                if self.state.get("account_failed", False):
                    await asyncio.sleep(3600) 
                    continue


                # 1. DAILY RESET CHECK
                current_date = datetime.now(timezone.utc).strftime('%Y-%m-%d')
                if current_date != self.state.get("last_utc_date"):
                    self.state["daily_start_balance"] = self.state["balance"]
                    self.state["last_utc_date"] = current_date
                    self.state["daily_halted"] = False 
                    self.save_state()
                    logging.info(f"New UTC Day. Daily Start Balance: ${self.state['balance']:.2f}")


                # 2. FETCH DATA
                self.latest_market = await self.analyzer.get_market_data()
                if not self.latest_market:
                    await asyncio.sleep(10)
                    continue


                # 3. NEWS CRAWLER
                current_hour = datetime.now(timezone.utc).hour
                if current_hour != self.last_news_hour:
                    headlines, self.macro = await self.analyzer.get_macro_sentiment()
                    for h in headlines: self.brain.log_headline(h, "Crawler")
                    self.last_news_hour = current_hour


                # 4. EQUITY GUARDIAN CIRCUIT BREAKERS
                current_price = self.latest_market["price"]
                floating_pnl = 0.0
                if self.state["position"] != 0:
                    trade = self.state["active_trade"]
                    if trade["side"] == "LONG":
                        floating_pnl = (current_price - trade["entry_price"]) * trade["amount_btc"]
                    else:
                        floating_pnl = (trade["entry_price"] - current_price) * trade["amount_btc"]


                current_equity = self.state["balance"] + floating_pnl
                total_dd = (INITIAL_BALANCE - current_equity) / INITIAL_BALANCE
                daily_dd = (self.state["daily_start_balance"] - current_equity) / self.state["daily_start_balance"]


                if total_dd >= MAX_TOTAL_DRAWDOWN_PCT:
                    logging.critical("🚨 TOTAL MAX DRAWDOWN HIT! PERMANENT HALT.")
                    if self.state["position"] != 0:
                        self.execute_auto_close(current_price, "CIRCUIT_BREAKER_TOTAL_DD")
                    self.state["account_failed"] = True
                    self.save_state()
                    await self.tg.send_alert("🚨 *ACCOUNT FAILED* 🚨\nTotal Drawdown hit 3%.\nBot permanently halted.")
                    continue


                if daily_dd >= DAILY_MAX_DRAWDOWN_PCT:
                    if not self.state["daily_halted"]:
                        logging.warning("⚠️ DAILY MAX DRAWDOWN HIT! HALTING FOR THE DAY.")
                        if self.state["position"] != 0:
                            self.execute_auto_close(current_price, "CIRCUIT_BREAKER_DAILY_DD")
                        self.state["daily_halted"] = True
                        self.save_state()
                        await self.tg.send_alert("⚠️ *DAILY DRAWDOWN HIT* ⚠️\nBot halted until midnight UTC.")
                    await asyncio.sleep(60)
                    continue


                atr = self.latest_market["atr"]
                ema50 = self.latest_market["ema50"]
                rsi = self.latest_market["rsi"]
                ob = self.latest_market["active_ob"]
                
                # 5. MANAGE OPEN TRADE
                if self.state["position"] != 0:
                    trade = self.state["active_trade"]
                    if (trade["side"] == "LONG" and current_price <= trade["stop_loss"]) or (trade["side"] == "SHORT" and current_price >= trade["stop_loss"]):
                        self.execute_auto_close(trade["stop_loss"], "STOP_LOSS")
                    elif (trade["side"] == "LONG" and current_price >= trade["take_profit"]) or (trade["side"] == "SHORT" and current_price <= trade["take_profit"]):
                        self.execute_auto_close(trade["take_profit"], "TAKE_PROFIT")
                    else:
                        if trade["side"] == "LONG" and current_price > trade["highest_price"]:
                            trade["highest_price"] = current_price
                            if (trade["highest_price"] - trade["entry_price"]) > (atr * TRAIL_ACTIVATION_ATR):
                                new_sl = trade["highest_price"] - (atr * TRAIL_DISTANCE_ATR)
                                if new_sl > trade["stop_loss"]:
                                    trade["stop_loss"] = new_sl
                                    self.save_state()
                                    asyncio.create_task(self.tg.send_alert(f"📈 Trail Moved UP: ${new_sl:,.2f}"))
                        elif trade["side"] == "SHORT" and current_price < trade["lowest_price"]:
                            trade["lowest_price"] = current_price
                            if (trade["entry_price"] - trade["lowest_price"]) > (atr * TRAIL_ACTIVATION_ATR):
                                new_sl = trade["lowest_price"] + (atr * TRAIL_DISTANCE_ATR)
                                if new_sl < trade["stop_loss"]:
                                    trade["stop_loss"] = new_sl
                                    self.save_state()
                                    asyncio.create_task(self.tg.send_alert(f"📉 Trail Moved DOWN: ${new_sl:,.2f}"))


                # 6. LOOK FOR ENTRY
                elif self.state["position"] == 0:
                    # BLACK SWAN SAFETY GUARD
                    if self.macro.get("global_risk") == "CRITICAL_BLACK_SWAN":
                        self.brain.log_skipped_trade("GLOBAL_BLACK_SWAN_DETECTED", self.latest_market, self.current_l2_imb)
                        await asyncio.sleep(60)
                        continue


                    side = None
                    skip_reason = None
                    at_ob = ob and (abs(current_price - ob) / current_price < OB_DISTANCE_PCT)
                    
                    if at_ob:
                        if self.current_l2_imb > L2_IMBALANCE_LONG and rsi < RSI_OVERSOLD and current_price > ema50:
                            side = "LONG"
                        elif self.current_l2_imb < L2_IMBALANCE_SHORT and rsi > RSI_OVERBOUGHT and current_price < ema50:
                            side = "SHORT"
                        else:
                            skip_reason = "OB_HIT_NO_CONFLUENCE"
                    else:
                        if abs(self.current_l2_imb) > 0.75:
                             skip_reason = "L2_SPIKE_NOT_AT_OB"


                    if skip_reason:
                        self.brain.log_skipped_trade(skip_reason, self.latest_market, self.current_l2_imb)


                    if side:
                        if self.state["balance"] >= PROFIT_BUFFER_THRESHOLD: risk = BUFFER_RISK_PCT
                        elif self.state["consecutive_wins"] >= 2: risk = STREAK_RISK_PCT
                        else: risk = BASE_RISK_PCT


                        stop_dist = atr * ATR_STOP_MULT
                        amount_btc = (self.state["balance"] * risk) / stop_dist
                        if amount_btc * current_price > (self.state["balance"] * 0.95):
                            amount_btc = (self.state["balance"] * 0.95) / current_price


                        entry_price = current_price * (1 + SLIPPAGE_PCT) if side == "LONG" else current_price * (1 - SLIPPAGE_PCT)
                        stop_loss = entry_price - stop_dist if side == "LONG" else entry_price + stop_dist
                        take_profit = entry_price + (stop_dist * ATR_TP_MULT) if side == "LONG" else entry_price - (stop_dist * ATR_TP_MULT)


                        self.state["active_trade"] = {
                            "side": side, "entry_price": entry_price, "amount_btc": amount_btc,
                            "stop_loss": stop_loss, "take_profit": take_profit,
                            "trailing_active": False, "highest_price": entry_price, "lowest_price": entry_price,
                            "atr": atr, "risk_pct": risk, "l2_imb": self.current_l2_imb,
                            "rsi": rsi, "ob": ob
                        }
                        self.state["position"] = 1 if side == "LONG" else -1
                        self.save_state()


                        asyncio.create_task(self.tg.send_alert(
                            f"🚨 *CONFLUENCE ENTRY: {side}* 🚨\n"
                            f"• L2: {self.current_l2_imb:+.2f} | RSI: {rsi:.1f}\n"
                            f"• Order Block: {ob}\n"
                            f"• Entry: ${entry_price:,.2f}\n"
                            f"• SL: ${stop_loss:,.2f} | TP: ${take_profit:,.2f}"
                        ))


                await asyncio.sleep(60)
            except Exception as e:
                logging.error(f"Strategy Error: {e}")
                await asyncio.sleep(10)


    def execute_auto_close(self, exit_price, reason):
        trade = self.state["active_trade"]
        actual_exit = exit_price * (1 - SLIPPAGE_PCT) if trade["side"] == "LONG" else exit_price * (1 + SLIPPAGE_PCT)
        entry_fee = (trade["entry_price"] * trade["amount_btc"]) * PROPR_TAKER_FEE
        exit_fee = (actual_exit * trade["amount_btc"]) * PROPR_TAKER_FEE
        
        gross_pnl = (actual_exit - trade["entry_price"]) * trade["amount_btc"] if trade["side"] == "LONG" else (trade["entry_price"] - actual_exit) * trade["amount_btc"]
        net_pnl = gross_pnl - entry_fee - exit_fee
        
        self.state["balance"] += net_pnl
        self.state["consecutive_wins"] = self.state["consecutive_wins"] + 1 if net_pnl > 0 else 0
        self.state["position"] = 0
        self.state["active_trade"] = {}
        self.save_state()
        
        self.brain.log_trade(trade, actual_exit, net_pnl, self.state["balance"], reason, self.macro)
        
        emoji = "✅" if net_pnl > 0 else "🛑"
        asyncio.create_task(self.tg.send_alert(
            f"{emoji} *CLOSED {trade['side']}* {emoji}\n"
            f"• Reason: {reason}\n"
            f"• PnL: ${net_pnl:+.2f}\n"
            f"• Balance: ${self.state['balance']:.2f}"
        ))


    async def main(self):
        await asyncio.gather(
            self.tg.start_async(),
            self.hyperliquid_ws_loop(),
            self.strategy_loop()
        )
